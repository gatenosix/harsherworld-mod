# Chance for children to get sick each year

random_yearly_everyone_pulse {
    on_actions = {
        hw_child_ill_pulse
        hw_smallpox_pulse
    }
}

hw_child_ill_pulse = {
    trigger = {
        is_child = yes
        NOT = { has_trait = ill }
        NOT = { has_trait = pneumonic }
        NOT = { age > 11 }
    }

    random_events = {
        chance_to_happen = 60

        # If the above check succeeds, the below will determine if no event happens anyway
        chance_of_no_event = {
            value = 0

            if = {
                limit = {
                    age <=2
                }

                add = 10
            }

            else_if = {
                limit = {
                    age <=4
                }

                add = 15
            }

            else_if = {
                limit = {
                    age = 5
                }

                add = 20
            }

            else_if = {
                limit = {
                    age = 6
                }

                add = 30
            }

            else_if = {
                limit = {
                    age = 7
                }

                add = 35
            }

            else_if = {
                limit = {
                    age = 8
                }

                add = 45
            }

            else_if = {
                limit = {
                    age = 9
                }

                add = 55
            }

            else_if = {
                limit = {
                    age = 10
                }

                add = 65
            }

            else {
                add = 75
            }

            if = {
                limit = {
                    has_trait = sickly
                }

                multiply = 0.7
            }
        }

        200 = health.1001 # Ill trait
        50 = health.1002 # Pneumonia trait
    }
}

hw_smallpox_pulse = { # PROBLEM: Seems to run fine from the 'run' debug command, but when trying as part of the mod as an on_action, it does not.
	if = {
		limit = {
			location ?= {
				any_province_epidemic = {
					epidemic_type = epidemic_type:smallpox
					count > 0
				}
			}
		}
	}
	
	trigger_event = {
		id = health.1006
	}
}
